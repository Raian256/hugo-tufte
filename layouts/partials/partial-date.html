{{/* Parses a date that may be only partly known: "YYYY", "YYYY-MM" or "YYYY-MM-DD" (an unquoted TOML date works too), or missing altogether. Used by the book and album logs. Usage: partial "partial-date.html" (dict "value" .finished "what" "books: Title") Returns a dict: date for sorting: the first day of the year or month when that's all that's known, plus a second per level of precision, so that on the same day "2026-01-01" > "2026-01" > "2026"; less precise dates therefore sort last in their year or month; undated entries sort after everything datetime the date as written, for ("" if undated) precision "year", "month", "day" or "none" when as precisely as known: "5 March 2024", "March 2024", "2024" whenInYear without the year: "5 March", "March", "" year the year heading it belongs under: "2024", or "Undated" month the month heading it belongs under: "March 2024", "2024" when only the year is known, or "Undated" day within its month: "Tuesday 5", or "" */}} {{ $raw := "" }} {{ with .value }}{{ $raw = substr (printf "%v" . | strings.TrimSpace) 0 10 }}{{ end }} {{ if and $raw (not (findRE `^\d{4}(-\d{2}(-\d{2})?)?$` $raw)) }} {{ errorf "%s: date must be YYYY, YYYY-MM or YYYY-MM-DD, not %q" .what $raw }} {{ end }} {{ $r := dict "datetime" $raw }} {{ if not $raw }} {{ $r = merge $r (dict "date" (time.AsTime "0001-01-01") "precision" "none" "when" "" "whenInYear" "" "year" "Undated" "month" "Undated" "day" "") }} {{ else if eq (len $raw) 4 }} {{ $date := time.AsTime (print $raw "-01-01") }} {{ $r = merge $r (dict "date" $date "precision" "year" "when" $raw "whenInYear" "" "year" $raw "month" $raw "day" "") }} {{ else if eq (len $raw) 7 }} {{ $date := time.AsTime (print $raw "-01") }} {{ $r = merge $r (dict "date" ($date.Add (time.ParseDuration "1s")) "precision" "month" "when" ($date.Format "January 2006") "whenInYear" ($date.Format "January") "year" ($date.Format "2006") "month" ($date.Format "January 2006") "day" "") }} {{ else }} {{ $date := time.AsTime $raw }} {{ $r = merge $r (dict "date" ($date.Add (time.ParseDuration "2s")) "precision" "day" "when" ($date.Format "2 January 2006") "whenInYear" ($date.Format "2 January") "year" ($date.Format "2006") "month" ($date.Format "January 2006") "day" ($date.Format "Monday 2")) }} {{ end }} {{ return $r }}
