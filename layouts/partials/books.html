{{/* Reads data/books.toml and returns its entries newest first, with the entry's `authors` joined into `by` for display, `stars` rendered and `post` resolved to a published page (or nil if it doesn't exist or is still a draft). `finished` may be "YYYY", "YYYY-MM" or "YYYY-MM-DD" (see partial-date.html). (dict "by" (delimit .authors ", " " & ") "finished" $d.date "datetime" $d.datetime "when" $d.when "whenInYear" $d.whenInYear "yearHeading" $d.year "stars" (partial "stars.html" .rating) "page" $post )) }} {{ end }} {{ return sort $books "finished" "desc" }}
