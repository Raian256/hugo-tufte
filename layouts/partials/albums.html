{{/* Reads data/albums.toml and returns its entries newest first, with `title`, `artist` and `year` filled in, `listened` parsed as a date, `stars` rendered, `page` resolved to a published post (or nil), and `cover` as a resized image resource (or nil). Nothing is downloaded here. Details and covers come from files saved by scripts/sync-albums.py: data/musicbrainz.json and assets/covers/. Values written in the data file take precedence. */}} {{ $saved := hugo.Data.musicbrainz | default dict }} {{ $albums := slice }} {{ range hugo.Data.albums.albums }} {{ $mbid := "" }} {{ with .mbid }}{{ with findRE `[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}` . 1 }}{{ $mbid = index . 0 }}{{ end }}{{ end }} {{ $details := index $saved $mbid | default dict }} {{ $title := .title | default $details.title }} {{ $artist := .artist | default $details.artist }} {{ $year := .year | default $details.year }} {{ if not (and $title $artist) }} {{ warnf "albums: %q isn't saved yet; run `python3 themes/hugo-tufte/scripts/sync-albums.py` from the site root" (.mbid | default "an entry") }} {{ continue }} {{ end }} {{ $post := "" }} {{ with .post }} {{ with site.GetPage . }}{{ $post = . }}{{ else }}{{ warnf "albums: no published page at %q" . }}{{ end }} {{ end }} {{/* Cover file name: a hash of `cover` if given, else the MBID. Must match sync-albums.py. */}} {{ $key := $mbid }} {{ with .cover }}{{ $key = substr (sha1 .) 0 16 }}{{ end }} {{ $cover := "" }} {{ with $key }} {{ with resources.GetMatch (printf "covers/%s.*" .) }}{{ $cover = .Fill "180x180 webp q80" }}{{ end }} {{ end }} {{ $albums = $albums | append (merge . (dict "mbid" $mbid "title" $title "artist" $artist "year" $year "listened" (time.AsTime .listened) "stars" (partial "stars.html" .rating) "page" $post "cover" $cover )) }} {{ end }} {{ return sort $albums "listened" "desc" }}
